cmake_minimum_required(VERSION 3.0.2)
project(falcon_native)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs sensor_msgs)
find_package(gazebo REQUIRED)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS} ${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
add_library(native_contact_observer SHARED src/contact_observer.cpp)
target_compile_features(native_contact_observer PRIVATE cxx_std_14)
target_link_libraries(native_contact_observer ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
add_library(native_raw_depth_observer SHARED src/raw_depth_observer.cpp)
target_compile_features(native_raw_depth_observer PRIVATE cxx_std_14)
target_link_libraries(native_raw_depth_observer ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
add_library(native_depth_self_filter SHARED src/depth_self_filter.cpp)
target_compile_features(native_depth_self_filter PRIVATE cxx_std_14)
target_link_libraries(native_depth_self_filter ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES})
